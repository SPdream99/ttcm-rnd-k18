import { GetActiveNodesUseCase } from '@/core/use-cases/GetActiveNodes';
import { MockDataNodeRepo } from '@/infrastructure/repositories/MockDataNodeRepo';

export default async function Home() {
  // Dependency Injection: Lắp ráp các thành phần lại với nhau
  const repository = new MockDataNodeRepo(); 
  const useCase = new GetActiveNodesUseCase(repository);

  // Thực thi nghiệp vụ
  const activeNodes = await useCase.execute();

  return (
    <main className="p-8">
      <h1 className="text-2xl font-bold mb-4">Network Data Nodes</h1>
      <ul>
        {activeNodes.map((node) => (
          <li key={node.id} className="mb-2 p-4 border rounded">
            IP: {node.ipAddress} - Bandwidth: {node.bandwidth} Mbps
          </li>
        ))}
      </ul>
    </main>
  );
}
