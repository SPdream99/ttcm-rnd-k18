// Cách cập nhật lên Firebase:
// 1. Mở Firebase Console → Project "e-learning-466a0"
// 2. Vào Firestore Database → Tab "Rules" (Quy tắc)
// 3. Xoá toàn bộ nội dung cũ và dán toàn bộ đoạn code dưới đây vào
// 4. Bấm nút "Publish" (Xuất bản)
// ==============================================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ==========================================
    // HELPER FUNCTIONS
    // ==========================================
    
    function isSignedIn() {
      return request.auth != null;
    }
    
    function isOwner(userId) {
      return isSignedIn() && request.auth.uid == userId;
    }

    function getUserData() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data;
    }

    function isAdmin() {
      return isSignedIn() && (
        getUserData().role == 'admin' || 
        getUserData().role == 'school'
      );
    }

    function isTeacher() {
      return isSignedIn() && 
        (getUserData().role == 'teacher' || getUserData().role == 'admin');
    }

    // ==========================================
    // COLLECTION RULES
    // ==========================================

    // 1. Users Collection
    match /users/{userId} {
      allow read: if true;
      allow create: if true;
      allow update: if isOwner(userId) || isAdmin() || isSignedIn();
      allow delete: if isAdmin();
    }

    // 2. Shop Items Collection
    match /shop_items/{itemId} {
      allow read: if true;
      allow write: if isAdmin() || isTeacher();
    }

    // 3. Courses Collection
    match /courses/{courseId} {
      allow read: if true;
      allow create, update: if isTeacher() || isAdmin() || isSignedIn();
      allow delete: if isAdmin() || isSignedIn();
    }

    // 4. Learning Path Collection
    match /learning_path/{lpathId} {
      allow read: if true;
      allow create, update: if isTeacher() || isAdmin() || isSignedIn();
      allow delete: if isAdmin() || isSignedIn();
    }

    // 5. Resources Collection
    match /resources/{resourceId} {
      allow read: if true;
      allow create, update, delete: if isTeacher() || isAdmin() || isSignedIn();
    }

    // 6. Game Info Collection
    match /game_info/{gameId} {
      allow read: if true;
      allow create, update: if isTeacher() || isAdmin() || isSignedIn();
      allow delete: if isAdmin() || isSignedIn();
    }

    // 7. Game Results Collection
    match /game_results/{resultId} {
      allow read: if true;
      allow create, update: if true;
      allow delete: if isAdmin();
    }

    // 8. Enrollments Collection
    match /enrollments/{enrollmentId} {
      allow read: if true;
      allow create, update: if isSignedIn() || true;
      allow delete: if isAdmin();
    }

  }
}
